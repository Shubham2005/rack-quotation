  const handleSharePDF = async () => {
    if (!quote || quote.enrichedCart.length === 0) {
      alert("Please add items to the quote first.");
      return;
    }

    setIsGeneratingPDF(true); // Triggers the UI to show the letterhead temporarily

    try {
      // 1. Wait a tiny bit for React to un-hide the letterhead
      await new Promise((res) => setTimeout(res, 300));

      const element = letterheadRef.current;
      if (!element) throw new Error("Letterhead element not found");

      // 2. Capture the full letterhead at high resolution
      const canvas = await html2canvas(element, {
        backgroundColor: "#ffffff",
        scale: 2,
        useCORS: true,
        imageTimeout: 0,
        logging: false,
      });

      // 3. Build a multi-page A4 PDF by slicing the canvas
      const pdf = new jsPDF({ orientation: "portrait", unit: "mm", format: "a4" });
      const pageW = pdf.internal.pageSize.getWidth();   // 210 mm
      const pageH = pdf.internal.pageSize.getHeight();  // 297 mm

      // How many canvas pixels correspond to one A4 page height
      const pxPerMm = canvas.width / pageW;
      const pxPerPage = pageH * pxPerMm;
      const totalPages = Math.ceil(canvas.height / pxPerPage);

      for (let page = 0; page < totalPages; page++) {
        if (page > 0) pdf.addPage();

        const srcY = page * pxPerPage;
        const srcH = Math.min(pxPerPage, canvas.height - srcY);

        // Slice just this page's portion from the full canvas
        const sliceCanvas = document.createElement("canvas");
        sliceCanvas.width = canvas.width;
        sliceCanvas.height = srcH;
        sliceCanvas.getContext("2d").drawImage(canvas, 0, -srcY);

        // JPEG 0.92 â€” good quality; white invoice compresses well
        const sliceData = sliceCanvas.toDataURL("image/jpeg", 0.92);
        const sliceH = srcH / pxPerMm; // mm height of this slice
        pdf.addImage(sliceData, "JPEG", 0, 0, pageW, sliceH);
      }

      // 4. Share or download
      const fileName = `VARUN_Quotation_${customerName ? customerName.replace(/\s+/g, "_") : "Estimate"}.pdf`;
      const pdfBlob = pdf.output("blob");
      const file = new File([pdfBlob], fileName, { type: "application/pdf" });

      // 5. Trigger Native Share (Mobile / Supported Browsers)
      if (navigator.canShare && navigator.canShare({ files: [file] })) {
        await navigator.share({
          title: "VARUN Enterprise Quotation",
          text: "Please find the attached quotation.",
          files: [file],
        });
      } else {
        // Fallback for Desktop: Force Download
        pdf.save(fileName);
      }
    } catch (err) {
      console.error(err);
      alert("Something went wrong while generating the PDF.");
    } finally {
      setIsGeneratingPDF(false); // Hide the letterhead again
    }
  };
